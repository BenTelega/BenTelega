### 6. `src/utils/telegram.js`
```javascript
export function getTelegramUser() {
  const tg = window.Telegram?.WebApp
  if (tg?.initDataUnsafe?.user) {
    return tg.initDataUnsafe.user
  }
  return null
}

export function getTelegramInitData() {
  const tg = window.Telegram?.WebApp
  return tg?.initData || ''
}

export function shareToChat(videoUrl) {
  const tg = window.Telegram?.WebApp
  if (tg) {
    tg.shareUrl(videoUrl)
  }
}

export function enableMainButton(text, onClick) {
  const tg = window.Telegram?.WebApp
  if (tg?.MainButton) {
    tg.MainButton.setText(text)
    tg.MainButton.onClick(onClick)
    tg.MainButton.show()
  }
}

export function disableMainButton() {
  const tg = window.Telegram?.WebApp
  if (tg?.MainButton) {
    tg.MainButton.hide()
    tg.MainButton.offClick()
  }
}
```

import { useState, useRef } from 'react'

export default function PhotoUpload({ onUpload }) {
  const [dragActive, setDragActive] = useState(false)
  const inputRef = useRef(null)

  const handleFile = (file) => {
    if (file && file.type.startsWith('image/')) {
      const reader = new FileReader()
      reader.onload = (e) => {
        onUpload(file, e.target.result)
      }
      reader.readAsDataURL(file)
    }
  }

  const handleDrag = (e) => {
    e.preventDefault()
    e.stopPropagation()
    if (e.type === 'dragenter' || e.type === 'dragover') {
      setDragActive(true)
    } else if (e.type === 'dragleave') {
      setDragActive(false)
    }
  }

  const handleDrop = (e) => {
    e.preventDefault()
    e.stopPropagation()
    setDragActive(false)
    if (e.dataTransfer.files && e.dataTransfer.files[0]) {
      handleFile(e.dataTransfer.files[0])
    }
  }

  const handleChange = (e) => {
    e.preventDefault()
    if (e.target.files && e.target.files[0]) {
      handleFile(e.target.files[0])
    }
  }

  const handleButtonClick = () => {
    inputRef.current?.click()
  }

  return (
    <div className="space-y-4">
      <div
        className={`relative border-2 border-dashed rounded-2xl p-8 text-center transition ${
          dragActive 
            ? 'border-purple-500 bg-purple-500/10' 
            : 'border-gray-600 bg-gray-800/50'
        }`}
        onDragEnter={handleDrag}
        onDragLeave={handleDrag}
        onDragOver={handleDrag}
        onDrop={handleDrop}
      >
        <input
          ref={inputRef}
          type="file"
          accept="image/*"
          className="hidden"
          onChange={handleChange}
        />
        
        <div className="text-6xl mb-4">📸</div>
        <h3 className="text-xl font-semibold mb-2">Загрузи фото</h3>
        <p className="text-gray-400 text-sm mb-4">
          Лицо должно быть хорошо видно
        </p>
        
        <button
          onClick={handleButtonClick}
          className="px-6 py-3 bg-purple-600 hover:bg-purple-700 rounded-xl font-semibold transition"
        >
          Выбрать из галереи
        </button>
        
        <p className="text-xs text-gray-500 mt-4">
          или перетащи фото сюда
        </p>
      </div>

      <div className="bg-gray-800/50 rounded-xl p-4">
        <h4 className="font-semibold mb-2">💡 Советы для лучшего результата:</h4>
        <ul className="text-sm text-gray-400 space-y-1">
          <li>✓ Лицо должно быть крупным</li>
          <li>✓ Хорошее освещение</li>
          <li>✓ Без очков и масок</li>
          <li>✓ Анфас или небольшой поворот</li>
        </ul>
      </div>
    </div>
  )
}

### 9. `src/components/GenerationProgress.jsx`
```jsx
export default function GenerationProgress({ status }) {
  const getProgressText = () => {
    switch(status.status) {
      case 'queued': return 'В очереди...'
      case 'processing': return 'Генерируем магию... ✨'
      case 'completed': return 'Готово! 🎉'
      case 'failed': return 'Ошибка 😕'
      default: return 'Загрузка...'
    }
  }

  const getProgressPercent = () => {
    if (status.progress) return status.progress
    if (status.status === 'queued') return 10
    if (status.status === 'processing') return 50
    if (status.status === 'completed') return 100
    return 0
  }

  return (
    <div className="text-center py-12">
      <div className="text-6xl mb-4 animate-pulse">🎬</div>
      
      <h3 className="text-xl font-semibold mb-2">
        {getProgressText()}
      </h3>
      
      <div className="w-full bg-gray-700 rounded-full h-4 mb-4 overflow-hidden">
        <div 
          className="bg-gradient-to-r from-purple-500 to-pink-500 h-4 rounded-full transition-all duration-500"
          style={{ width: `${getProgressPercent()}%` }}
        />
      </div>
      
      <p className="text-gray-400 text-sm">
        Обычно занимает 15-30 секунд
      </p>

      {status.estimated_time && (
        <p className="text-gray-500 text-xs mt-2">
          Осталось: ~{status.estimated_time} сек
        </p>
      )}
    </div>
  )
}

### 11. `src/App.jsx`
```jsx
import { useState, useEffect } from 'react'
import PhotoUpload from './components/PhotoUpload'
import TemplateGrid from './components/TemplateGrid'
import GenerationProgress from './components/GenerationProgress'
import VideoPreview from './components/VideoPreview'
import { getTelegramUser, enableMainButton, disableMainButton, shareToChat } from './utils/telegram'

// Заглушка для API (временно)
const API_URL = import.meta.env.VITE_API_URL || 'https://dev-api.memeface.app'

function App() {
  const [step, setStep] = useState('upload')
  const [photo, setPhoto] = useState(null)
  const [photoPreview, setPhotoPreview] = useState(null)
  const [selectedTemplate, setSelectedTemplate] = useState(null)
  const [generationStatus, setGenerationStatus] = useState(null)
  const [videoUrl, setVideoUrl] = useState(null)
  const [user, setUser] = useState(null)
  const [freeGenerations, setFreeGenerations] = useState(2)

  useEffect(() => {
    const tgUser = getTelegramUser()
    setUser(tgUser)
  }, [])

  const handlePhotoUpload = (file, preview) => {
    setPhoto(file)
    setPhotoPreview(preview)
    setStep('select')
  }

  const handleTemplateSelect = (template) => {
    setSelectedTemplate(template)
  }

  const handleGenerate = async () => {
    if (!photo || !selectedTemplate) return

    if (freeGenerations <= 0) {
      alert('Лимит генераций исчерпан. Покупка за Telegram Stars будет добавлена в следующей версии.')
      return
    }

    setStep('generating')
    disableMainButton()

    // Заглушка для API
    setTimeout(() => {
      setVideoUrl('https://commondatastorage.googleapis.com/gtv-videos-bucket/sample/ForBiggerJoyrides.mp4') // ЗАМЕНИТЬ НА РЕАЛЬНОЕ ВИДЕО
      setStep('preview')
      setFreeGenerations(prev => prev - 1)
    }, 3000) // Симуляция генерации 3 секунды
  }

  const handleShare = () => {
    if (videoUrl) {
      shareToChat(videoUrl)
    }
  }

  const handleNewVideo = () => {
    setStep('upload')
    setPhoto(null)
    setPhotoPreview(null)
    setSelectedTemplate(null)
    setVideoUrl(null)
    setGenerationStatus(null)
  }

  // Показывать MainButton на разных шагах
  useEffect(() => {
    if (step === 'select' && selectedTemplate) {
      enableMainButton('Сгенерировать 🎬', handleGenerate)
    } else if (step === 'preview') {
      enableMainButton('Поделиться 📤', handleShare)
    } else {
      disableMainButton()
    }

    return () => disableMainButton()
  }, [step, selectedTemplate])

  return (
    <div className="min-h-screen bg-gradient-to-b from-purple-900 to-black text-white p-4">
      <header className="text-center mb-6">
        <h1 className="text-3xl font-bold bg-gradient-to-r from-pink-500 to-purple-500 bg-clip-text text-transparent">
          🎭 MemeFace
        </h1>
        <p className="text-gray-400 text-sm mt-1">Твоё лицо в мемах</p>
        {user && (
          <div className="mt-2 text-xs text-gray-500">
            👤 {user.first_name} {user.last_name || ''}
          </div>
        )}
        <div className="mt-2 text-xs text-yellow-400">
          ⭐ Осталось генераций: {freeGenerations}
        </div>
      </header>

      <main className="max-w-md mx-auto">
        {step === 'upload' && (
          <PhotoUpload onUpload={handlePhotoUpload} />
        )}

        {step === 'select' && photoPreview && (
          <>
            <div className="mb-4">
              <img 
                src={photoPreview} 
                alt="Your photo" 
                className="w-24 h-24 rounded-full mx-auto border-4 border-purple-500 object-cover"
              />
            </div>
            <TemplateGrid 
              selectedTemplate={selectedTemplate}
              onSelect={handleTemplateSelect}
            />
          </>
        )}

        {step === 'generating' && (
          <GenerationProgress status={{ status: 'processing', progress: 75, estimated_time: 15 }} />
        )}

        {step === 'preview' && videoUrl && (
          <VideoPreview 
            videoUrl={videoUrl}
            onNewVideo={handleNewVideo}
          />
        )}
      </main>

      {step === 'preview' && (
        <div className="max-w-md mx-auto mt-4">
          <button
            onClick={handleNewVideo}
            className="w-full py-3 bg-gray-800 hover:bg-gray-700 rounded-xl font-semibold transition"
          >
            🔄 Создать новое видео
          </button>
        </div>
      )}

      <footer className="text-center mt-8 text-xs text-gray-600">
        <p>🔒 Фото удаляются через 24 часа</p>
        <p>© 2026 MemeFace</p>
      </footer>
    </div>
  )
}

export default App
